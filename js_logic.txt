function generatePages() {
  const bookWrap = document.getElementById('bookWrap');
  
  // İçindekiler sayfasını oluştur
  const indexContent = document.getElementById('indexContent');
  indexContent.innerHTML = '';
  
  // Alfabetik sıraya göre
  const sortedContribs = [...contributors].sort((a, b) => a.name.localeCompare(b.name));
  
  sortedContribs.forEach((contrib, idx) => {
    const pageNum = 3 + idx; // Sayfa 3'ten başla (0=kapak, 1=önsöz, 2=içindekiler)
    const firstLetter = contrib.name[0].toUpperCase();
    
    const indexItem = document.createElement('div');
    indexItem.className = 'index-item';
    const pageNumForThisPage = pageNum;
    indexItem.onclick = () => goTo(pageNumForThisPage);
    indexItem.innerHTML = `<span class="idx-alpha">${firstLetter}</span><span class="idx-name">${contrib.name}</span><span class="idx-dots"></span><span class="idx-pg">${pageNum}</span>`;
    indexContent.appendChild(indexItem);
  });
  
  // Her kişi için bir sayfa oluştur
  sortedContribs.forEach((contrib, idx) => {
    const pageNum = 3 + idx;
    
    let photo1HTML = `<svg class="mem-photo-icon" viewBox="0 0 40 40" fill="none" xmlns="http://www.w3.org/2000/svg"><rect x="3" y="8" width="34" height="24" rx="2.5" stroke="#4a3f30" stroke-width="1.4"/><circle cx="20" cy="20" r="6.5" stroke="#4a3f30" stroke-width="1.4"/><path d="M14 8l2-3h8l2 3" stroke="#4a3f30" stroke-width="1.4"/><circle cx="31" cy="13" r="1.5" fill="#4a3f30" opacity="0.4"/></svg><span class="mem-photo-label">Fotoğraf buraya eklenecek</span>`;
    
    const memPage = document.createElement('div');
    memPage.className = 'page';
    memPage.id = `mem-${idx}`;
    
    let blocksHTML = `<div class="mem-block"><div class="mem-photo"><div class="mem-photo-inner">${photo1HTML}</div></div><p class="mem-text">${contrib.text}</p><div class="mem-meta"><span class="mem-author">${contrib.name}</span></div></div>`;
    
    memPage.innerHTML = `<div class="mem-page"><div class="mem-body">${blocksHTML}</div><div class="mem-footer"><span class="mem-ftr-title">Geleceği Yakala</span><span class="mem-ftr-num">${pageNum}</span></div></div>`;
    bookWrap.appendChild(memPage);
  });
  
  pages = document.querySelectorAll('.page');
  total = pages.length;
  updateCounter();
}

function goTo(pageNum) {
  if (pageNum < 0 || pageNum >= total) return;
  
  pages.forEach((p, i) => {
    p.classList.toggle('active', i === pageNum);
  });
  
  cur = pageNum;
  updateCounter();
}

function next() {
  goTo(cur + 1);
}

function prev() {
  goTo(cur - 1);
}

function updateCounter() {
  const counter = document.getElementById('counter');
  counter.textContent = `${cur + 1} / ${total}`;
}

// Sayfa tuşları
document.addEventListener('keydown', (e) => {
  if (e.key === 'ArrowRight' || e.key === ' ') next();
  if (e.key === 'ArrowLeft') prev();
  if (e.key === 'PageDown') next();
  if (e.key === 'PageUp') prev();
});
